"""Harmless actual CPU ownership/deadline probe; never imports a model library."""
import argparse
import time
from pathlib import Path
import illustration_core as boundary
import owned_native

parser=argparse.ArgumentParser()
parser.add_argument("--out",type=Path,required=True)
parser.add_argument("--sleep",type=float,default=0)
args=parser.parse_args()
ownership,actual=owned_native.await_ownership(args.out/"ownership.json",boundary.read)
boundary.fresh(args.out/"identity.json",actual)
end=time.monotonic()+args.sleep
while time.monotonic()<end and not (args.out/"cancel.json").exists():
    time.sleep(.01)
