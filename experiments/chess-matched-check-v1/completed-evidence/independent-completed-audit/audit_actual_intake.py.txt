def digest(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(65536), b""): h.update(block)
    return h.hexdigest()


def inside(root, name):
    if type(name) is not str or Path(name).is_absolute() or re.match(r"^[A-Za-z]:", name):
        raise ValueError("relative evidence path required")
    target = (root / name).resolve()
    if not target.is_relative_to(root.resolve()): raise ValueError("evidence escapes root")
    return target


class Reader:
    def __init__(self): self.before = {}
    def bind(self, path, sha=None, size=None):
        path = Path(path).resolve()
        item = {"bytes": path.stat().st_size, "sha256": digest(path)}
        if sha is not None: assert item["sha256"] == sha, str(path)
        if size is not None: assert item["bytes"] == size, str(path)
        if str(path) in self.before: assert self.before[str(path)] == item, str(path)
        self.before[str(path)] = item
        return path
    def read(self, path):
        path = self.bind(path)
        return json.loads(path.read_text(encoding="utf-8"))
    def stable(self):
        for name, item in self.before.items():
            path = Path(name)
            assert path.stat().st_size == item["bytes"] and digest(path) == item["sha256"], name


def key(fen): return ref.board_key(fen)


def add_manifest(manifest, games, boards):
    for item in manifest["items"]:
        games.add(ref.canonical_game(item["game_id"]))
        boards.add(key(item["fen"]))
        for ply in item.get("legal_replay", []):
            before, after = chess.Board(ply["before"]), chess.Board(ply["after"])
            assert before.is_valid() and after.is_valid()
            # Independently verify that the retained transition is a legal ply,
            # without trusting a source's annotation spelling/field naming.
            legal_matches = 0
            for move in before.legal_moves:
                pushed = before.copy(stack=False); pushed.push(move)
                legal_matches += pushed.fen() == after.fen()
            assert legal_matches == 1
            boards.update((key(before.fen()), key(after.fen())))


def old_rows(path):
    import zstandard
    with path.open("rb") as raw:
        with io.TextIOWrapper(zstandard.ZstdDecompressor().stream_reader(raw), encoding="utf-8") as stream:
            yield from csv.DictReader(stream)


def fact_roles(row, games, boards, successors=False):
    before = chess.Board(row["source_before_fen"])
    assert before.is_valid()
    setup = chess.Move.from_uci(row["setup_uci"])
    assert setup and setup in before.legal_moves
    boards.add(key(before.fen())); before.push(setup)
    assert before.fen() == row["fen"] and before.is_valid()
    games.add(ref.canonical_game(row["game_id"]))
    boards.add(key(before.fen()))
    if successors:
        move = chess.Move.from_uci(row["selected_uci"])
        assert move in before.legal_moves
        before.push(move); boards.add(key(before.fen()))


def retained_successors(repo, rd):
    import numpy as np
    directory = repo / "data/private/gothamchess/constructed-middlegame-eval-v1"
    seal = rd.read(directory / "raw-seal.json")
    assert seal["planned"] == 6 and seal["evaluator_opened"] is False
    for name, sha in seal["files_sha256"].items(): rd.bind(inside(directory, name), sha)
    raw = rd.read(directory / "raw-results.json")["rows"][:4]
    closed = rd.read(directory / "move-raw-closed.json")
    assert closed["rows"] == raw and closed["forbid_further_move_calls"] is True
    inputs = rd.read(directory / "move-input.json")["positions"]
    assert len(inputs) == 1 and len(raw) == 4
    board = chess.Board(inputs[0]["fen"])
    assert board.is_valid()
    legal = {m.uci() for m in board.legal_moves}
    choices = set(); occurrences = 0
    for row in raw:
        assert row["status"] == "completed"
        if "classifier" not in row["condition"]:
            assert row["parsed"] is True and row["raw_legal"] is True
            assert row["raw_uci"] in legal
            choices.add(row["raw_uci"]); occurrences += 1
            continue
        artifact = row["logit_artifact"]
        path = rd.bind(inside(directory, artifact["path"]), artifact["sha256"])
        scores = np.load(path, allow_pickle=False)
        labels = row["id2label"]
        assert scores.dtype == np.float32 and scores.shape == (1968,) and np.isfinite(scores).all()
        assert len(labels) == len(set(labels)) == 1968
        ranked = sorted(range(1968), key=lambda i: (-float(scores[i]), i))
        masked = [i for i in ranked if labels[i] in legal]
        truth = row["raw"]
        assert truth["raw_choice_uci"] == labels[ranked[0]]
        assert truth["legal_choice_uci"] == labels[masked[0]]
        assert truth["raw_choice_is_legal"] is (labels[ranked[0]] in legal)
        assert truth["legal_move_count"] == len(legal)
        for name, indices in (("raw_topk", ranked[:5]), ("legal_topk", masked[:5])):
            assert len(truth[name]) == len(indices)
            for item, index in zip(truth[name], indices, strict=True):
                assert item == {"class_id": index, "move_uci": labels[index],
                                "logit": float(scores[index]), "legal": labels[index] in legal}
                choices.add(labels[index]); occurrences += 1
        choices.update((labels[ranked[0]], labels[masked[0]])); occurrences += 2
    result = set()
    for uci in choices:
        move = chess.Move.from_uci(uci)
        if move in board.legal_moves:
            after = board.copy(stack=False); after.push(move); result.add(key(after.fen()))
    assert len(result) == 7 and occurrences == 26
    return result, {"distinct_legal_moves": sorted(choices & legal), "top5_and_first_occurrences": occurrences}

def physical_rows(path):
    import pyarrow.parquet as pq
    parquet = pq.ParquetFile(path)
    ordinal = 0
    for group in range(parquet.num_row_groups):
        offset = 0
        for batch in parquet.iter_batches(batch_size=4096, row_groups=[group],
                columns=["PuzzleId", "GameId", "FEN", "Moves"], use_threads=False):
            for row in batch.to_pylist():
                yield {"source_ordinal":ordinal, "row_group":group, "row_in_group":offset}, row
                ordinal += 1; offset += 1


def filter_counts(record, unavailable):
    board = chess.Board(record["fen"]); kept = []; counts = Counter()
    for move in board.legal_moves:
        if move.promotion or board.is_castling(move) or board.is_en_passant(move): counts["special_move"] += 1; continue
        successor = board.copy(stack=False); successor.push(move)
        next_key = key(successor.fen())
        if not successor.is_valid() or next_key in unavailable or next_key in (record["source_before_key"], record["setup_after_key"]):
            counts["successor_overlap"] += 1; continue
        facts = ref.predicate_facts(record["fen"], move.uci())
        kept.append((move.from_square, facts, next_key))
    for a, b in combinations(kept, 2):
        matched = a[0] == b[0] and all(a[1][f] == b[1][f] for f in ("moving_piece", "moving_colour", "is_capture", "captured_piece"))
        if not matched or a[1]["gives_check"] == b[1]["gives_check"]: counts["unmatched_pair"] += 1
        elif a[2] == b[2]: counts["successor_pair_collision"] += 1
    return counts


def select_independently(path, known_games, known_keys):
    locations, records = [], []
    def rows():
        for location, row in islice(physical_rows(path), 200000):
            locations.append(location); records.append(row)
            yield row
    selected = ref.reconstruct_fixture_selection(rows(), known_games, known_keys, target=32, cap=200000)
    assert selected["status"] == "complete" and len(selected["pairs"]) == 32
    assert len(locations) == len(selected["trace"])
    filters, games, keys = Counter(), set(known_games), set(known_keys)
    trace = []
    for index, (row, outcome) in enumerate(zip(records, selected["trace"], strict=True)):
        assert outcome["source_ordinal"] == index
        trace.append({**locations[index], "outcome":outcome["terminal"]})
        if outcome["terminal"] in ("accepted", "no_legal_matched_pair_after_all_filters"):
            record = ref.parse_row(row); filters.update(filter_counts(record, keys))
        if outcome["terminal"] == "accepted":
            pair = next(p for p in selected["pairs"] if p["source_ordinal"] == index)
            games.add(pair["game_id"]); keys.update((pair["source_before_key"], pair["setup_after_key"], *pair["successor_keys"]))
    for pair in selected["pairs"]: pair.update(locations[pair["source_ordinal"]])
    return selected["pairs"], trace, dict(filters)
