def resource_counts(records, stage):
    results = {"samples":len(records), "minimum_physical_available_bytes":None,
        "minimum_commit_headroom_bytes":None, "minimum_gpu_free_bytes":None,"max_recorded_monotonic_gap_seconds":0.}
    previous=None
    for sample in records:
        keys=("physical_total_bytes","physical_available_bytes","commit_limit_bytes","commit_total_bytes","gpu_total_bytes","gpu_free_bytes")
        assert all(type(sample[k]) is int and sample[k]>=0 for k in keys)
        assert 0 < sample["physical_total_bytes"] >= sample["physical_available_bytes"]
        assert 0 < sample["commit_limit_bytes"] >= sample["commit_total_bytes"]
        assert 0 < sample["gpu_total_bytes"] >= sample["gpu_free_bytes"]
        values=(sample["physical_available_bytes"], sample["commit_limit_bytes"]-sample["commit_total_bytes"], sample["gpu_free_bytes"])
        for value, threshold, key in zip(values, (10,16,13) if stage=="start" else (4,8,1),
                ("minimum_physical_available_bytes","minimum_commit_headroom_bytes","minimum_gpu_free_bytes"), strict=True):
            assert value >= threshold*1024**3
            results[key]=value if results[key] is None else min(results[key],value)
        if "monotonic" in sample:
            now=sample["monotonic"]
            assert type(now) in (int,float) and math.isfinite(now)
            if previous is not None:
                gap=now-previous; assert 0<=gap<=3
                results["max_recorded_monotonic_gap_seconds"]=max(results["max_recorded_monotonic_gap_seconds"],gap)
            previous=now
    return results


def inventory(root, reader):
    return {p.relative_to(root).as_posix():{"bytes":p.stat().st_size,"sha256":digest(reader.bind(p))}
            for p in sorted(root.rglob("*")) if p.is_file()}

def main():

    args=parser.parse_args()
    if not args.completed_run_audit_authorized: parser.error("wait for terminal run and explicit root audit GO")
    workspace=Path(__file__).resolve().parents[2]
    out=workspace/"outputs/chess-matched-check-completed-independent-audit.json"
    if out.exists(): parser.error("refusing existing audit output")
    run(workspace,out)
