def packet():
    rows=[{"case_id":f"mc-pair-{index:02d}-{side}","fen":"7k/8/8/8/8/8/8/K1R5 w - - 0 1",
           "selected_uci":"c1c8" if side==0 else "c1c2"} for index in range(32) for side in range(2)]
    def rank(row):
        data=json.dumps({"seed":4082032,"case_id":row["case_id"]},sort_keys=True,separators=(",",":"),ensure_ascii=True).encode()
        return hashlib.sha256(b"matched-check-order-v1\n"+data).hexdigest(),row["case_id"]
    return {"schema":"chess.matched-check.inputs.v1","rows":sorted(rows,key=rank)}


def labels_for(rows):
    return [{"case_id":row["case_id"],"expected":core.expected_facts(row)} for row in rows]


def seal_fixture(tmp_path,status=None):
    rows=packet()["rows"]
    labels=labels_for(rows)
    raw=[{"case_id":row["case_id"],"status":"completed","text":json.dumps(label["expected"])}
         for row,label in zip(rows,labels,strict=True)]
    if status:
        status(raw,labels)
    path=tmp_path/"labels.json"
    boundary.fresh(path,{"rows":labels})
    common.seal_raw(tmp_path/"raw",raw,[row["case_id"] for row in rows],{})
    return path,raw

def test_invalid_unknown_missing_are_not_false_predictions(tmp_path):

        true=[i for i,label in enumerate(labels) if label["expected"]["gives_check"]]

            rows[index].update(status=status,text="{}")

        "schema_failure":1,"truncated":1,"error":1,"not_run":1}
    assert score["counts"]["planned"]==64

def test_explicit_lora_exclusion_and_original_byte_order():
    import numpy as np
    class Tensor:
        def __init__(self,a):self.a=a
        def numel(self):return self.a.size
        def detach(self):return self
        def contiguous(self):return self
        def view(self,dtype):return Tensor(self.a.view(dtype))
        def reshape(self,*shape):return Tensor(self.a.reshape(*shape))
        def __getitem__(self,key):return Tensor(self.a[key])
        def cpu(self):
            assert self.a.nbytes<=1048576
            return self
        def numpy(self):return self.a
    values=[("base.z",Tensor(np.arange(600000,dtype=np.uint32))),
            ("base.lora_A.default.weight",Tensor(np.array([7],dtype=np.uint32))),
            ("base.a",Tensor(np.array([42],dtype=np.uint32)))]
    model=types.SimpleNamespace(named_parameters=lambda:iter(values))
    torch=types.SimpleNamespace(uint8=np.uint8)
    result=worker.state_digest(model,torch)
    expected=hashlib.sha256()
    for name,tensor in (values[0],values[2]):
        expected.update(name.encode());expected.update(tensor.a.tobytes())
    assert result=={"sha256":expected.hexdigest(),"tensors":2,"scalars":600001}
    assert worker.state_digest(model,torch,adapter=True)["scalars"]==1


def test_actual_owned_cpu_identity_and_target_only_cleanup(tmp_path):
    executable,environment,identity=owned_native.configuration(REPO)
    control=subprocess.Popen([executable,"-B","-c","import time;time.sleep(30)"],env=environment)
    child=None
    try:
        child=subprocess.Popen([executable,"-B",str(HERE/"identity_probe_worker.py"),"--out",str(tmp_path),"--sleep","30"],env=environment)
        boundary.fresh(tmp_path/"ownership.json",{"pid":child.pid,"native_launch_identity":identity})
        deadline=time.monotonic()+5
        while not (tmp_path/"identity.json").exists() and time.monotonic()<deadline:time.sleep(.01)
        actual=boundary.read(tmp_path/"identity.json")
        assert actual["pid"]==child.pid and actual["torch_imported"] is False
        with pytest.raises(RuntimeError,match="deadline"):
            runner.wait_owned(child,tmp_path,"cpu",time.monotonic()-600,healthy,resources)
        assert runner.stop(child,lambda:boundary.fresh(tmp_path/"cancel.json",{"fixture":True})) is False
        assert child.poll()==0 and control.poll() is None
        boundary.fresh(tmp_path/"probe-receipt.json",{"actual_popen_pid":child.pid,
            "actual_worker_pid":actual["pid"],"child_exit_code":child.poll(),
            "unrelated_control_survived":True,"model_imported":False,"model_calls":0})
    finally:
        for process in (child,control):
            if process is not None and process.poll() is None:
                process.terminate();process.wait(timeout=5)
