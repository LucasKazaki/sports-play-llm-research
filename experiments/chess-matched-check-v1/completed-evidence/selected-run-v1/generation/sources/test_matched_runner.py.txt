def packet():
    rows=[{"case_id":f"mc-pair-{index:02d}-{side}","fen":"7k/8/8/8/8/8/8/K1R5 w - - 0 1",
           "selected_uci":"c1c8" if side==0 else "c1c2"} for index in range(32) for side in range(2)]
    def rank(row):
        data=json.dumps({"seed":4082032,"case_id":row["case_id"]},sort_keys=True,separators=(",",":"),ensure_ascii=True).encode()
        return hashlib.sha256(b"matched-check-order-v1\n"+data).hexdigest(),row["case_id"]
    return {"schema":"chess.matched-check.inputs.v1","rows":sorted(rows,key=rank)}


def labels_for(rows):
    return [{"case_id":row["case_id"],"expected":core.expected_facts(row["fen"],row["selected_uci"])} for row in rows]

def test_explicit_lora_exclusion_and_original_byte_order():
    import numpy as np
    class Tensor:
        def __init__(self,a):self.a=a
        def numel(self):return self.a.size
        def detach(self):return self
        def contiguous(self):return self
        def view(self,dtype):return Tensor(self.a.view(dtype))
        def reshape(self,*shape):return Tensor(self.a.reshape(*shape))
        def __getitem__(self,key):return Tensor(self.a[key])
        def cpu(self):
            assert self.a.nbytes<=1048576
            return self
        def numpy(self):return self.a
    values=[("base.z",Tensor(np.arange(600000,dtype=np.uint32))),
            ("base.lora_A.default.weight",Tensor(np.array([7],dtype=np.uint32))),
            ("base.a",Tensor(np.array([42],dtype=np.uint32)))]
    model=types.SimpleNamespace(named_parameters=lambda:iter(values))
    torch=types.SimpleNamespace(uint8=np.uint8)
    result=worker.state_digest(model,torch)
    expected=hashlib.sha256()
    for name,tensor in (values[0],values[2]):
        expected.update(name.encode());expected.update(tensor.a.tobytes())
    assert result=={"sha256":expected.hexdigest(),"tensors":2,"scalars":600001}
    assert worker.state_digest(model,torch,adapter=True)["scalars"]==1

class Clock:
    def __init__(self):self.now=0.
    def __call__(self):return self.now
    def sleep(self,seconds):self.now+=seconds


def parent_fixture(tmp_path,monkeypatch,*,mode="pass",clock=None):
    monkeypatch.setattr(runner,"protected_output",lambda repo,cohort,out:Path(out))
    cohort=tmp_path/"cohort";cohort.mkdir()
    payload=packet(); rows=payload["rows"]
    boundary.fresh(cohort/"inputs.json",payload)
    boundary.fresh(cohort/"labels.json",{"rows":labels_for(rows)})
    manifest={"files_sha256":{"inputs.json":boundary.sha(cohort/"inputs.json"),
                             "labels.json":boundary.sha(cohort/"labels.json")}}
    boundary.fresh(cohort/"frozen-data.json",manifest)
    extension={"source_sha256":{name:boundary.sha(HERE/name) for name in runner.SOURCES},
               "frozen_data_sha256":boundary.sha(cohort/"frozen-data.json")}
    monkeypatch.setattr(runner,"verify",lambda *args:(extension,manifest,payload,{}))
    monkeypatch.setattr(owned_native,"configuration",lambda repo:("fixture.exe",{},{}))
    launched=[]
    class Child:
        pid=987654
        def poll(self):return 0
    def popen(command,**kwargs):
        launched.append(command)
        out=tmp_path/"attempt";generation=out/"generation"
        if len(launched)==1:
            if mode=="partial_raw":
                (generation/"raw").mkdir()
                (generation/"raw/raw.json").write_bytes(b'{"partial":')
                (generation/"journal.jsonl").write_text(json.dumps({"case_id":rows[0]["case_id"],
                    "status":"completed","text":"{}"})+'\n')
                boundary.fresh(generation/"prepared-01.json",{"case_id":rows[1]["case_id"]})
                return Child()
            raw=[{"case_id":row["case_id"],"status":"completed","text":json.dumps(label["expected"])}
                 for row,label in zip(rows,labels_for(rows),strict=True)]
            common.seal_raw(generation/"raw",raw,[row["case_id"] for row in rows],{})
            if mode!="missing_outcome":
                boundary.fresh(generation/"worker-outcome.json",{"planned":64,"status":"completed_raw_accounting",
