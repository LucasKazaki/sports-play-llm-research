"""Read-only retained-artifact audit. Never invokes a model, engine or backend.

The actual-run entry point requires explicit post-completion authorization and
completed worker/supervisor outcomes before opening any evaluator label file.
Pure helpers are exercised against finite fixtures before a real run completes.
"""
from collections import Counter
from datetime import datetime, timezone
import argparse
import hashlib
import json
import math
from pathlib import Path

FIELDS = ('moving_piece','moving_colour','is_capture','captured_piece','gives_check',
          'is_castling','is_en_passant','promotion_piece')
BOOLS = {'is_capture','gives_check','is_castling','is_en_passant'}
STEPS = [0,128,256,512,1024]


def require(condition, message):
    if not condition: raise ValueError(message)


def read(path): return json.loads(Path(path).read_text(encoding='utf-8'))


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(65536), b''): digest.update(block)
    return digest.hexdigest()


def finite(value): return type(value) in (int,float) and math.isfinite(value)


def bound_file(root, relative):
    result = (Path(root)/relative).resolve()
    result.relative_to(Path(root).resolve())
    return result


def verify_files(root, mapping):
    for name,digest in mapping.items():
        require(sha(bound_file(root,name))==digest,'file hash mismatch: '+name)


def check_updates(updates, permutation, target_counts, batch_size=8, expected_updates=1024):
    n = batch_size*expected_updates
    require(len(target_counts)==n,'training target count denominator')
    require(len(permutation)==n and all(type(i) is int for i in permutation)
            and sorted(permutation)==list(range(n)),'permutation must use every example once')
    require(len(updates)==expected_updates,'all planned optimizer updates required')
    for index,row in enumerate(updates):
        step=index+1
        require(type(row['step']) is int and row['step']==step,'optimizer step order/gap')
        require(finite(row['loss']) and row['loss']>=0,'nonfinite/negative loss')
        require(finite(row['gradient_norm_before_clip']) and row['gradient_norm_before_clip']>=0,'nonfinite gradient norm')
        require(finite(row['lr']) and math.isclose(row['lr'],2e-4*min(1.,step/32),rel_tol=1e-14,abs_tol=1e-18),'LR schedule differs')
        batch=permutation[index*batch_size:step*batch_size]
        require(type(row['target_tokens']) is int and row['target_tokens']==sum(target_counts[i] for i in batch),'batch token count differs')
        require(finite(row['elapsed_seconds']) and row['elapsed_seconds']>=0,'invalid update latency')
        for name in ('cuda_allocated_peak','cuda_reserved_peak'):
            require(type(row[name]) is int and row[name]>=0,'invalid CUDA peak counter')
            if index: require(row[name]>=updates[index-1][name],'CUDA peak counter went backwards')
        require(row['cuda_reserved_peak']>=row['cuda_allocated_peak'],'CUDA reserved peak below allocated peak')
    return {'updates':len(updates),'examples_consumed_once':n,'target_tokens':sum(target_counts),
        'loss_first':updates[0]['loss'],'loss_last':updates[-1]['loss'],
        'loss_min':min(r['loss'] for r in updates),'loss_max':max(r['loss'] for r in updates)}


def strict_parse(text, schema):
    def pairs(items):
        result={}
        for key,value in items:
            require(key not in result,'duplicate response key')
            result[key]=value
        return result
    def bad_constant(value): raise ValueError('nonfinite JSON constant')
    value=json.loads(text,object_pairs_hook=pairs,parse_constant=bad_constant)
    require(type(value) is dict and set(value)==set(FIELDS),'response must have exact eight fields')
    for field,item in value.items():
        if field in BOOLS:
            require(type(item) is bool or type(item) is str and item=='unknown','strict Boolean type')
        else:
            require(item is None or type(item) is str,'strict categorical type')
            require(item in schema['properties'][field]['enum'],'categorical value outside schema')
    return value


def independent_score(rows,labels,schema):
    require([r['case_id'] for r in rows]==[r['case_id'] for r in labels],'scoring case order differs')
    fields={field:{'correct':0,'unknown':0,'classes':{},'confusion':{}} for field in FIELDS}
    counts={'planned':len(rows),'all_eight_correct':0,'strict_parse_completed':0,
        'parse_error':0,'truncated':0,'error':0,'not_run':0}
    outcomes=[]
    for row,label in zip(rows,labels,strict=True):
        prediction=None
        if row['status']=='completed':
            try:
                prediction=strict_parse(row['text'],schema)
                counts['strict_parse_completed']+=1
            except (ValueError,TypeError): counts['parse_error']+=1
        else:
            require(row['status'] in ('truncated','error','not_run'),'invalid raw status')
            counts[row['status']]+=1
        correct={}
        for field in FIELDS:
            expected=label['expected'][field]
            actual=prediction[field] if prediction is not None else '__invalid__'
            match=type(expected) is type(actual) and expected==actual and actual!='unknown'
            correct[field]=match
            item=fields[field]
            item['correct']+=int(match)
            item['unknown']+=int(actual=='unknown')
            key=json.dumps(expected)
            category=item['classes'].setdefault(key,{'support':0,'correct':0})
            category['support']+=1
            category['correct']+=int(match)
            confusion=key+' -> '+json.dumps(actual)
            item['confusion'][confusion]=item['confusion'].get(confusion,0)+1
        complete=all(correct.values())
        counts['all_eight_correct']+=int(complete)
        outcomes.append({'case_id':row['case_id'],'all_eight_correct':complete,'field_correct':correct})
    return {'counts':counts,'fields':fields,'outcomes':outcomes}


def read_sealed_raw(directory,ids):
    directory=Path(directory)
    seal=read(directory/'seal.json')
    require(sha(directory/'raw.json')==seal['raw_sha256'],'raw seal mismatch before label join')
    raw=read(directory/'raw.json')
    require(seal['planned']==len(ids) and seal['planned_ids']==ids,'raw seal planned denominator/order')
    require(raw['planned']==len(ids) and [r['case_id'] for r in raw['rows']]==ids,'raw full denominator/order')
    require(raw['metadata']==seal['metadata'],'raw/seal metadata differs')
    return raw


def verify_all_raw_before_labels(run,splits):
    result={}
    for step in STEPS:
        name='dev-'+str(step)
        result[name]=read_sealed_raw(Path(run)/name,[r['case_id'] for r in splits['dev']])
    for condition in ('base','selected'):
        name='holdout-'+condition
        result[name]=read_sealed_raw(Path(run)/name,[r['case_id'] for r in splits['holdout']])
    return result


def check_raw_tokens(rows,inputs,tokenizer,core,model_vocab_size):
    require(type(model_vocab_size) is int and model_vocab_size>0,'invalid pinned model vocabulary dimension')
    require([r['case_id'] for r in rows]==[r['case_id'] for r in inputs],'raw/input identity order')
    status=Counter()
    for row,source in zip(rows,inputs,strict=True):
        status[row['status']]+=1
        if row['status'] in ('not_run','error'): continue
        prompt=core.build_prompt(source,'fen_uci_piece_map')
        ids=tokenizer.apply_chat_template([{'role':'user','content':prompt}],tokenize=True,
            add_generation_prompt=True,enable_thinking=False,return_dict=False)
        require(ids==row['prompt_ids'],'retained native prompt IDs differ')
        require(hashlib.sha256(str(ids).encode('utf-8')).hexdigest()==row['prompt_sha256'],'prompt ID hash differs')
        generated=row['new_ids']
        require(type(generated) is list and 1<=len(generated)<=128 and
            all(type(i) is int and 0<=i<model_vocab_size for i in generated),'invalid retained generated IDs')
        require(finite(row['elapsed_seconds']) and row['elapsed_seconds']>=0,'invalid generation latency')
        if tokenizer.eos_token_id in generated:
            end=generated.index(tokenizer.eos_token_id)
            require(row['status']=='completed','EOS/status mismatch')
            require(all(i==tokenizer.pad_token_id for i in generated[end+1:]),'tokens after generated EOS')
            text=tokenizer.decode(generated[:end],skip_special_tokens=False)
        else:
            require(row['status']=='truncated' and len(generated)==128,'non-EOS output must be cap truncation')
            text=tokenizer.decode(generated,skip_special_tokens=False)
        require(text==row['text'],'retained raw token/text decode differs')
    return dict(status)


def transition_truth(fen,uci):
    import chess
    board=chess.Board(fen)
    move=chess.Move.from_uci(uci)
    require(board.is_valid() and not board.chess960 and move in board.legal_moves,'invalid orthodox input/selected legal move')
    actor=board.piece_at(move.from_square)
    opponents={sq:piece for sq,piece in board.piece_map().items() if piece.color!=actor.color}
    after=board.copy(stack=False)
    after.push(move)
    remaining={sq for sq,piece in after.piece_map().items() if piece.color!=actor.color}
    removed=set(opponents)-remaining
    require(len(removed)<=1,'invalid capture transition')
    victim=opponents[next(iter(removed))] if removed else None
    return {'moving_piece':chess.piece_name(actor.piece_type),'moving_colour':'white' if actor.color else 'black',
        'is_capture':bool(removed),'captured_piece':chess.piece_name(victim.piece_type) if victim else None,
        'gives_check':after.is_attacked_by(actor.color,after.king(not actor.color)),
        'is_castling':actor.piece_type==chess.KING and abs(chess.square_file(move.to_square)-chess.square_file(move.from_square))==2,
        'is_en_passant':actor.piece_type==chess.PAWN and chess.square_file(move.from_square)!=chess.square_file(move.to_square) and board.piece_at(move.to_square) is None,
        'promotion_piece':chess.piece_name(after.piece_at(move.to_square).piece_type) if len(uci)==5 else None}


def selected_step(scores):
    require([r['step'] for r in scores]==STEPS,'all dev checkpoints required')
    require(all(r['counts']['planned']==128 and r['counts']['error']==0 and r['counts']['not_run']==0 for r in scores),'incomplete dev cohort cannot select')
    return min(scores,key=lambda r:(-r['counts']['all_eight_correct'],-r['counts']['strict_parse_completed'],r['step']))['step']


def check_resource_sample(sample,stage):
    names=['physical_total_bytes','physical_available_bytes','commit_limit_bytes','commit_total_bytes','gpu_total_bytes','gpu_free_bytes']
    require(all(type(sample.get(n)) is int and sample[n]>=0 for n in names),'resource counter missing/noninteger')
    require(0<sample['physical_total_bytes'] and sample['physical_available_bytes']<=sample['physical_total_bytes'],'physical counters invalid')
    require(0<sample['commit_limit_bytes'] and sample['commit_total_bytes']<=sample['commit_limit_bytes'],'commit counters invalid')
    require(0<sample['gpu_total_bytes'] and sample['gpu_free_bytes']<=sample['gpu_total_bytes'],'GPU counters invalid')
    values=[sample['physical_available_bytes'],sample['commit_limit_bytes']-sample['commit_total_bytes'],sample['gpu_free_bytes']]
    thresholds=[10,16,13] if stage=='start' else [4,8,1]
    require(all(value>=threshold*1024**3 for value,threshold in zip(values,thresholds)),'resource reserve breached')
    return values


def check_resources(preflight,records,ownership,supervision,phase):
    require(preflight['status']=='passed' and len(preflight['samples'])==10,'ten healthy preflight samples required')
    for sample in preflight['samples']: check_resource_sample(sample,'start')
    require(supervision['exit_code']==0 and not supervision['forced_owned_termination'] and not supervision['cancel_issues'],'owned run did not finish cleanly')
    require(supervision['pid']==ownership['pid'],'owned process identity differs')
    require(len(records)==supervision['sample_count'] and len(records)>0,'resource sample denominator')
    expected_policy={'start_physical_available_bytes':10*1024**3,
        'start_commit_headroom_bytes':16*1024**3,'start_gpu_free_bytes':13*1024**3,
        'stop_physical_available_bytes':4*1024**3,'stop_commit_headroom_bytes':8*1024**3,
        'stop_gpu_free_bytes':1024**3,'maximum_owned_request_seconds':5400.,
        'maximum_sample_gap_seconds':3.}
    require(ownership['policy']==expected_policy,'resource policy changed')
    last=ownership['started_monotonic']
    gaps=[]; values=[]
    for sample in records:
        stamp=sample['monotonic']
        require(finite(stamp) and stamp>=last,'nonmonotonic resource sequence')
        gaps.append(stamp-last)
        require(stamp-last<=3.,'resource sample gap breached')
        last=stamp
        values.append(check_resource_sample(sample,'running'))
    elapsed=supervision['elapsed_seconds']
    require(finite(elapsed) and 0<=elapsed<=5430.,'combined phase/cleanup wall exceeded')
    require(phase['phase']=='holdout' and finite(phase['started_monotonic']),'holdout phase boundary missing')
    start=ownership['started_monotonic']
    require(0<=phase['started_monotonic']-start<=3600,'training/dev phase exceeded')
    require(start+elapsed-phase['started_monotonic']<=1830.,'holdout phase/cleanup exceeded')
    return {'samples':len(records),'max_gap_seconds':max(gaps),
        'minimum_physical_available_bytes':min(v[0] for v in values),
        'minimum_commit_headroom_bytes':min(v[1] for v in values),
        'minimum_GPU_free_bytes':min(v[2] for v in values),'elapsed_seconds':elapsed,
        'recorded_training_dev_phase_seconds':phase['started_monotonic']-start,
        'recorded_holdout_plus_parent_finalization_seconds':start+elapsed-phase['started_monotonic'],
        'timing_limit':'Parent elapsed includes final file-only ledger/supervision writes; token/kernel stopping is cooperative and sampler-bounded, not instantaneous.'}


def native_target_count(tokenizer,row,expected,core):
    prompt=core.build_prompt(row,'fen_uci_piece_map')
    target=json.dumps(expected,sort_keys=True,separators=(',',':'))
    prefix=tokenizer.apply_chat_template([{'role':'user','content':prompt}],tokenize=True,
        add_generation_prompt=True,enable_thinking=False,return_dict=False)
    require(0<len(prefix)<=1408,'audit prompt length differs')
    joint=tokenizer.apply_chat_template([{'role':'user','content':prompt},{'role':'assistant','content':target}],
        tokenize=True,add_generation_prompt=False,enable_thinking=False,return_dict=False)
    require(joint[:len(prefix)]==prefix,'audit native prefix differs')
    count=joint.index(tokenizer.eos_token_id,len(prefix))+1-len(prefix)
    require(0<count<=127 and len(prefix)+count<=1536,'audit native supervised window differs')
    require(tokenizer.decode(joint[len(prefix):len(prefix)+count-1],skip_special_tokens=False)==target,'audit target roundtrip differs')
    return count


def adapter_shapes(config):
    hidden=config['hidden_size']; head=config['head_dim']; q=config['num_attention_heads']*head
    kv=config['num_key_value_heads']*head; intermediate=config['intermediate_size']
    dimensions={'q_proj':(hidden,q),'k_proj':(hidden,kv),'v_proj':(hidden,kv),
        'o_proj':(q,hidden),'gate_proj':(hidden,intermediate),'up_proj':(hidden,intermediate),
        'down_proj':(intermediate,hidden)}
    result={}
    for layer in range(config['num_hidden_layers']):
        for name,(in_dim,out_dim) in dimensions.items():
            module='self_attn' if name in ('q_proj','k_proj','v_proj','o_proj') else 'mlp'
            stem=f'base_model.model.model.layers.{layer}.{module}.{name}.'
            result[stem+'lora_A.weight']=(8,in_dim)
            result[stem+'lora_B.weight']=(out_dim,8)
    return result

        require(groups[0]['lr']==2e-4,'checkpoint optimizer LR differs')
        require(len(optimizer['state'])==(0 if step==0 else 392),'optimizer state denominator')
        for entry in optimizer['state'].values():
            require(float(entry['step'])==step and all(bool(torch.isfinite(entry[key]).all()) for key in ('exp_avg','exp_avg_sq')),'optimizer moments/steps invalid')
        checkpoints[step]=sha(folder/'manifest.json')
        del tensors,state
    dev=[]; raw_by_cohort={}; aggregate={}
    for step in STEPS:
        name='dev-'+str(step); raw=read_sealed_raw(run/name,[r['case_id'] for r in splits['dev']])

        require(raw['metadata']['inputs_sha256']==sha(cohort/'dev-inputs.json') and raw['metadata']['protocol_sha256']==sha(protocol),'dev input/protocol binding')

        raw_by_cohort[name]=raw['rows']; aggregate[name]=computed['counts']
    require(read(run/'dev-scores.json')==dev,'dev ranking source scores differ')
    selection=read(run/'selection.json'); winner=selected_step(dev)

    require(read(run/'phase.json')['selection_sha256']==sha(run/'selection.json'),'holdout phase selection binding')
    for condition in ('base','selected'):
        name='holdout-'+condition; raw=read_sealed_raw(run/name,[r['case_id'] for r in splits['holdout']])
        disabled=condition=='base' or winner==0
        require(raw['metadata']['selection_sha256']==sha(run/'selection.json') and raw['metadata']['inputs_sha256']==sha(cohort/'holdout-inputs.json'),'heldout selection/input binding')
        require(raw['metadata']['step']==(0 if disabled else winner) and raw['metadata']['adapter_disabled']==disabled,'heldout condition identity')

        require(not statuses.get('error') and not statuses.get('not_run'),'incomplete heldout generation')
        computed=independent_score(raw['rows'],labels['holdout'],plan['inputs']['schema'])
        reported=read(run/(name+'-score.json'))
        require(all(computed[key]==reported[key] for key in computed),'independent heldout score differs')
        require(reported['seal_sha256']==sha(run/name/'seal.json') and reported['labels_sha256']==frozen['files_sha256']['holdout-labels.json'],'heldout score provenance')
        raw_by_cohort[name]=raw['rows']; aggregate[name]=computed['counts']
    ledger=read(run/'outcome-ledger.json'); declared=read(run/'planned-outcomes.json')
    require(ledger['planned']==declared['planned']==1152 and ledger['cohorts']==raw_by_cohort,'full1152 outcome ledger differs from sealed raw')
    require(set(declared['cohorts'])==set(raw_by_cohort) and all(declared['cohorts'][key]==[r['case_id'] for r in rows] for key,rows in raw_by_cohort.items()),'declared full denominator identity')
    resources=check_resources(read(run/'preflight.json'),[json.loads(line) for line in (run/'resources.jsonl').read_text().splitlines()],ownership,supervision,read(run/'phase.json'))
