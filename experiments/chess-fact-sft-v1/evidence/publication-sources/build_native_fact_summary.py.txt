def read(path):
    return json.loads(path.read_text(encoding='utf-8'))

def sha(path):
    h=hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda:stream.read(65536),b''): h.update(block)
    return h.hexdigest()
