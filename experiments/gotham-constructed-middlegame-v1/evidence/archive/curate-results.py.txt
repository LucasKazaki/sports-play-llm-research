def sha(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(65536), b""):
            h.update(block)
    return h.hexdigest()


def read(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))


def create(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(value, stream, indent=2, allow_nan=False)
        stream.write("\n")


def verify_freeze():
    paths = subprocess.check_output(["git", "diff-tree", "--no-commit-id", "--name-only", "-r", FREEZE_COMMIT], cwd=REPO, text=True).splitlines()
    if len(paths) != 60: raise ValueError("pre-call freeze file denominator changed")
    for name in paths:
        data = subprocess.check_output(["git", "show", FREEZE_COMMIT + ":" + name], cwd=REPO)
        if hashlib.sha256(data).hexdigest() != sha(REPO / name):
            raise ValueError("pre-call artifact changed: " + name)
