def strict_facts(text):
    def unique(pairs):
        result={}
        for key,value in pairs:
            if key in result:raise ValueError('duplicate')
            result[key]=value
        return result
    def invalid(_):raise ValueError('nonfinite')
    obj=json.loads(text,object_pairs_hook=unique,parse_constant=invalid)
    if not isinstance(obj,dict) or set(obj)!=set(truth):raise ValueError('fields')
    for field in ('is_capture','gives_check','is_castling','is_en_passant'):
        if type(obj[field]) is not bool and obj[field]!='unknown':raise ValueError('boolean type')
    for field in ('moving_piece','captured_piece','promotion_piece'):
        allowed={'pawn','knight','bishop','rook','queen','king','unknown'}
        if field=='promotion_piece':allowed={'queen','rook','bishop','knight','unknown',None}
        elif field=='captured_piece':allowed.add(None)
        if obj[field] not in allowed:raise ValueError('piece enum')
    if obj['moving_colour'] not in ('white','black','unknown'):raise ValueError('colour')
    return obj

    except (ValueError,TypeError):pass
