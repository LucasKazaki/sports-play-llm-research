    paths = {}
    for name, row in (("move", move), ("fact", fact), ("evaluator", {"case_id": "fixture",
            "next_displayed_move_uci": "e2e4", "immediate_facts": truth})):
        path = tmp_path / (name + ".json")
        path.write_text(json.dumps({"positions": [row]}), encoding="utf-8")
        paths[name] = {"path": str(path), "sha256": subject.sha(path)}
    plan = {"conditions": {key: {} for key in subject.CONDITIONS},
            "planned_denominators": {"total_raw_condition_outcomes": 6},
            "input_files": {"move_policy": paths["move"], "immediate_facts": paths["fact"]},
            "evaluator_separation": paths["evaluator"]}
    out = tmp_path / "attempt"
    subject.initialize(out, plan, {}, source_bindings={})
    events = []
    mapping = adapter.expected_action_space()
    class FakeBackend:
        def __init__(self, task, condition): self.task, self.condition = task, condition
        def encode(self, row):
            assert set(row) == ({"case_id", "fen"} if self.task == "move" else {"case_id", "fen", "selected_uci"})
            events.append(("encode", self.task, self.condition, dict(row)))
            return {"input_ids": [1, 2, 3], "attention_mask": [1, 1, 1]}

            events.append(("predict", self.task, self.condition))
            if self.task == "fact":

            if self.condition in subject.MOVES[:2]:
                scores = np.zeros(1968, dtype=np.float32)
                scores[mapping.index("e2e4")] = 1
                return {"logits": scores, "id2label": mapping}
            return {"new_ids": [10], "text": "e2e4", "eos_found": False}
        def close(self): events.append(("close", self.task, self.condition))
    return out, plan, events, FakeBackend, paths


def test_move_outputs_immutably_close_before_fact_input_and_no_later_move(fixture):
    out, plan, events, backend, paths = fixture
    saw_fact = []
    def reader(path):
        if Path(path) == Path(paths["evaluator"]["path"]): pytest.fail("generator read evaluator")
        if Path(path) == Path(paths["fact"]["path"]):
            closed = subject.verify_move_close(out)
            assert [row["condition"] for row in closed["rows"]] == list(subject.MOVES)
            assert all(row["attempted"] for row in closed["rows"])
            assert len([e for e in events if e[0] == "predict"]) == 4
            saw_fact.append(True)
        return subject.read(path)
    def factory(task, condition):
        if saw_fact: assert task == "fact"
        return backend(task, condition)
    rows = subject.run_generation(out, factory, payload_reader=reader)
    assert saw_fact == [True]
    assert len(rows) == 6 and all(row["attempted"] for row in rows)
    assert sum(e[0] == "predict" for e in events) == 6
    assert len([e for e in events if e[:2] == ("encode", "fact")]) == 1
    subject.seal_all(out)
    result = subject.evaluate(out)
    assert result["planned"] == 6
    assert sum(x["matches_displayed_move"] for x in result["conditions"].values() if x["task"] == "move_policy") == 4
    assert sum(x["all_eight_correct"] for x in result["conditions"].values() if x["task"] == "given_move_immediate_facts") == 2
    assert "accuracy" not in result

def test_fact_decode_requires_native_eos_and_only_padding_after_it():
    class Tokenizer:
        eos_token_id, pad_token_id = 4, 0
        def decode(self, ids, **_): return str(ids)
    decode = subject.reviewed_modules()[0].decode_generated
    assert decode(Tokenizer(), [1, 4, 0])["status"] == "completed"
    assert decode(Tokenizer(), [1, 2])["status"] == "truncated"
    with pytest.raises(ValueError, match="nonpadding"): decode(Tokenizer(), [1, 4, 2])

def test_backend_factory_latches_against_later_move_request(monkeypatch):
    import local_backends
    marker = object()
    monkeypatch.setattr(local_backends, "FactBackend", lambda *_: marker)
    monkeypatch.setattr(local_backends, "MoveBackend", lambda *_: marker)
    factory = local_backends.Factory(Path.cwd(), {}, {}, lambda: False)
    assert factory("move", subject.MOVES[0]) is marker
    assert factory("fact", "fact_family") is marker
    with pytest.raises(ValueError, match="after fact"):
        factory("move", subject.MOVES[3])
