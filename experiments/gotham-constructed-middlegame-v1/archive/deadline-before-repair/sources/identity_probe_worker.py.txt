"""Bounded stdlib/metadata-only CPU probe for the new owned-native wrapper."""
import argparse
import json
from pathlib import Path
import time
import owned_native

parser = argparse.ArgumentParser()
parser.add_argument("--ownership", type=Path, required=True)
parser.add_argument("--result", type=Path, required=True)
parser.add_argument("--sleep", type=float, default=0)
args = parser.parse_args()
_, actual = owned_native.await_ownership(args.ownership,
    lambda path: json.loads(Path(path).read_text(encoding="utf-8")))
with args.result.with_suffix(".tmp").open("x", encoding="utf-8") as stream:
    json.dump(actual, stream)
args.result.with_suffix(".tmp").replace(args.result)
time.sleep(min(max(args.sleep, 0), 30))
